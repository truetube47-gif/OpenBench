class GGUFAnalyzer:
    async def analyze_from_hf(self, repo_id: str, filename: str = None):
        # Auto-find GGUF file if not specified
        if not filename:
            files = list_repo_files(repo_id)
            gguf_files = [f for f in files if f.endswith(".gguf")]
            filename = gguf_files[0] if gguf_files else None

        # Download only header (Range request) for metadata
        local_path = hf_hub_download(repo_id, filename, force_download=False)

        parser = GGUFParser(local_path)
        parser.parse()

        return {
            "repo_id": repo_id,
            "filename": filename,
            "metadata": parser.metadata,
            "architecture": parser.metadata.get("general.architecture"),
            "parameter_count": parser.metadata.get("general.parameter_count"),
            "context_length": parser.metadata.get("llama.context_length") or

            "quantization": self.detect_quant(parser),
            "tensor_count": len(parser.tensors),
